### 🟢 2.1 Factory Pattern

**Prioridade:** ALTA  
**Complexidade:** BAIXA  
**Impacto:** MÉDIO

**Problema Atual:**
- Criação de entidades complexas espalhada pelo código
- Lógica de inicialização duplicada
- Difícil garantir objetos sempre válidos

**Solução:**
Centralizar criação de entidades em factories.

**Implementação Sugerida:**

```typescript
// Domain Factory
export class QuartoFactory {
  static criar(dto: CadastrarQuartoDto): Quarto {
    // Validações de domínio
    this.validarCapacidade(dto.capacidade);
    this.validarPreco(dto.precoDiaria);
    
    // Criar entidade
    const quarto = new Quarto({
      id: uuid(),
      numero: dto.numero,
      capacidade: dto.capacidade,
      tipo: dto.tipo,
      precoDiaria: new Dinheiro(dto.precoDiaria, 'BRL'),
      status: StatusQuarto.LIVRE,
      frigobar: dto.frigobar,
      cafeManha: dto.cafeManha,
      arCondicionado: dto.arCondicionado,
      tv: dto.tv,
      criadoEm: new Date(),
      atualizadoEm: new Date()
    });
    
    // Adicionar camas
    dto.tiposCama.forEach(tipo => {
      quarto.adicionarCama(tipo);
    });
    
    return quarto;
  }
  
  static criarComCamasDefault(
    numero: string,
    tipo: TipoQuarto
  ): Quarto {
    // Lógica de criação com valores padrão por tipo
    const camasPadrao = {
      [TipoQuarto.BASICO]: [TipoCama.SOLTEIRO, TipoCama.SOLTEIRO],
      [TipoQuarto.MODERNO]: [TipoCama.CASAL_QUEEN],
      [TipoQuarto.LUXO]: [TipoCama.CASAL_KING]
    };
    
    return this.criar({
      numero,
      capacidade: camasPadrao[tipo].length * 2,
      tipo,
      precoDiaria: this.getPrecoPadrao(tipo),
      frigobar: tipo !== TipoQuarto.BASICO,
      cafeManha: true,
      arCondicionado: tipo !== TipoQuarto.BASICO,
      tv: true,
      tiposCama: camasPadrao[tipo]
    });
  }
  
  private static validarCapacidade(cap: number): void {
    if (cap <= 0) {
      throw new InvalidCapacidadeException(
        'Capacidade deve ser maior que zero'
      );
    }
  }
  
  private static validarPreco(preco: number): void {
    if (preco <= 0) {
      throw new InvalidPrecoException(
        'Preço deve ser maior que zero'
      );
    }
  }
  
  private static getPrecoPadrao(tipo: TipoQuarto): number {
    return {
      [TipoQuarto.BASICO]: 100,
      [TipoQuarto.MODERNO]: 200,
      [TipoQuarto.LUXO]: 400
    }[tipo];
  }
}
```

### 🟢 2.2 Specification Pattern

**Prioridade:** ALTA  
**Complexidade:** MÉDIA  
**Impacto:** ALTO

**Problema Atual:**
- Lógica de validação de disponibilidade espalhada
- Difícil combinar regras de negócio complexas
- Regras hardcoded nos services

**Solução:**
Encapsular regras de negócio em especificações reutilizáveis.

**Implementação Sugerida:**

```typescript
// Interface base
interface ISpecification<T> {
  isSatisfiedBy(candidate: T): boolean;
  and(other: ISpecification<T>): ISpecification<T>;
  or(other: ISpecification<T>): ISpecification<T>;
  not(): ISpecification<T>;
}

// Classe abstrata
abstract class Specification<T> implements ISpecification<T> {
  abstract isSatisfiedBy(candidate: T): boolean;
  
  and(other: ISpecification<T>): ISpecification<T> {
    return new AndSpecification(this, other);
  }
  
  or(other: ISpecification<T>): ISpecification<T> {
    return new OrSpecification(this, other);
  }
  
  not(): ISpecification<T> {
    return new NotSpecification(this);
  }
}

// Especificações concretas
class QuartoDisponivelSpecification extends Specification<Quarto> {
  isSatisfiedBy(quarto: Quarto): boolean {
    return quarto.status === StatusQuarto.LIVRE;
  }
}

class QuartoCapacidadeMinimaSpecification extends Specification<Quarto> {
  constructor(private capacidadeMinima: number) {
    super();
  }
  
  isSatisfiedBy(quarto: Quarto): boolean {
    return quarto.capacidade >= this.capacidadeMinima;
  }
}

class QuartoPrecoMaximoSpecification extends Specification<Quarto> {
  constructor(private precoMaximo: Dinheiro) {
    super();
  }
  
  isSatisfiedBy(quarto: Quarto): boolean {
    return quarto.precoDiaria.getValor() <= this.precoMaximo.getValor();
  }
}

class QuartoTemComodidadeSpecification extends Specification<Quarto> {
  constructor(private comodidade: keyof Pick<Quarto, 'frigobar' | 'cafeManha' | 'arCondicionado' | 'tv'>) {
    super();
  }
  
  isSatisfiedBy(quarto: Quarto): boolean {
    return quarto[this.comodidade] === true;
  }
}

// Combinadores
class AndSpecification<T> extends Specification<T> {
  constructor(
    private left: ISpecification<T>,
    private right: ISpecification<T>
  ) {
    super();
  }
  
  isSatisfiedBy(candidate: T): boolean {
    return this.left.isSatisfiedBy(candidate) && 
           this.right.isSatisfiedBy(candidate);
  }
}

class OrSpecification<T> extends Specification<T> {
  constructor(
    private left: ISpecification<T>,
    private right: ISpecification<T>
  ) {
    super();
  }
  
  isSatisfiedBy(candidate: T): boolean {
    return this.left.isSatisfiedBy(candidate) || 
           this.right.isSatisfiedBy(candidate);
  }
}

class NotSpecification<T> extends Specification<T> {
  constructor(private spec: ISpecification<T>) {
    super();
  }
  
  isSatisfiedBy(candidate: T): boolean {
    return !this.spec.isSatisfiedBy(candidate);
  }
}
```

**Uso no Service:**
```typescript
@Injectable()
export class QuartosService {
  async buscarQuartosParaReserva(
    capacidadeMinima: number,
    precoMaximo: number,
    precisaFrigobar: boolean
  ): Promise<Quarto[]> {
    // Compor especificações
    let spec: ISpecification<Quarto> = new QuartoDisponivelSpecification()
      .and(new QuartoCapacidadeMinimaSpecification(capacidadeMinima))
      .and(new QuartoPrecoMaximoSpecification(new Dinheiro(precoMaximo)));
    
    if (precisaFrigobar) {
      spec = spec.and(new QuartoTemComodidadeSpecification('frigobar'));
    }
    
    // Buscar todos os quartos
    const quartos = await this.repository.findAll();
    
    // Filtrar usando especificação
    return quartos.filter(q => spec.isSatisfiedBy(q));
  }
  
  async validarQuartoParaReserva(quartoId: string): Promise<boolean> {
    const quarto = await this.repository.findById(quartoId);
    
    if (!quarto) {
      throw new NotFoundException('Quarto não encontrado');
    }
    
    // Especificação composta
    const especificacao = new QuartoDisponivelSpecification()
      .and(
        new NotSpecification(
          new QuartoStatusSpecification(StatusQuarto.MANUTENCAO)
            .or(new QuartoStatusSpecification(StatusQuarto.LIMPEZA))
        )
      );
    
    return especificacao.isSatisfiedBy(quarto);
  }
}
```

**Benefícios:**
- ✅ Regras de negócio encapsuladas e reutilizáveis
- ✅ Fácil combinar regras complexas (AND, OR, NOT)
- ✅ Testável isoladamente
- ✅ Código declarativo e legível
- ✅ Single Responsibility (cada spec tem uma regra)

**Aplicação no Sistema:**
- Validação de disponibilidade de quartos (RF18)
- Filtros de busca avançada
- Regras de elegibilidade para promoções (futuro)

**Requisitos Atendidos:**
- RF18: Impedir reserva em quartos indisponíveis
- RNF13: Clean code
- RNF14: Manutenibilidade

### 🟢 2.3 Strategy Pattern

**Prioridade:** MÉDIA-ALTA  
**Complexidade:** MÉDIA  
**Impacto:** ALTO

**Problema Futuro:**
- Cálculo de preço pode variar (promoções, descontos, alta temporada)
- Regras de cancelamento podem mudar
- Políticas de pagamento diferentes

**Solução:**
Encapsular algoritmos intercambiáveis em strategies.

**Implementação Sugerida:**

```typescript
// Interface Strategy
interface ICalculoValorReservaStrategy {
  calcular(reserva: Reserva, quarto: Quarto): Dinheiro;
}

// Estratégia padrão
class CalculoValorPadraoStrategy implements ICalculoValorReservaStrategy {
  calcular(reserva: Reserva, quarto: Quarto): Dinheiro {
    const diarias = this.calcularDiarias(
      reserva.dataCheckIn,
      reserva.dataCheckOut
    );
    
    return quarto.precoDiaria.multiplicar(diarias);
  }
  
  private calcularDiarias(checkIn: Date, checkOut: Date): number {
    const diff = checkOut.getTime() - checkIn.getTime();
    return Math.ceil(diff / (1000 * 60 * 60 * 24));
  }
}

// Estratégia com desconto por longa permanência
class CalculoValorLongaPermanenciaStrategy 
  implements ICalculoValorReservaStrategy {
  
  calcular(reserva: Reserva, quarto: Quarto): Dinheiro {
    const diarias = this.calcularDiarias(
      reserva.dataCheckIn,
      reserva.dataCheckOut
    );
    
    let valorBase = quarto.precoDiaria.multiplicar(diarias);
    
    // Desconto progressivo
    if (diarias >= 30) {
      // 20% desconto para 30+ dias
      valorBase = valorBase.multiplicar(0.8);
    } else if (diarias >= 14) {
      // 15% desconto para 14-29 dias
      valorBase = valorBase.multiplicar(0.85);
    } else if (diarias >= 7) {
      // 10% desconto para 7-13 dias
      valorBase = valorBase.multiplicar(0.9);
    }
    
    return valorBase;
  }
  
  private calcularDiarias(checkIn: Date, checkOut: Date): number {
    const diff = checkOut.getTime() - checkIn.getTime();
    return Math.ceil(diff / (1000 * 60 * 60 * 24));
  }
}

// Estratégia para alta temporada
class CalculoValorAltaTemporadaStrategy 
  implements ICalculoValorReservaStrategy {
  
  constructor(private multiplicador: number = 1.5) {}
  
  calcular(reserva: Reserva, quarto: Quarto): Dinheiro {
    const diarias = this.calcularDiarias(
      reserva.dataCheckIn,
      reserva.dataCheckOut
    );
    
    // Aumenta preço em alta temporada
    const precoDiariaAjustado = quarto.precoDiaria.multiplicar(
      this.multiplicador
    );
    
    return precoDiariaAjustado.multiplicar(diarias);
  }
  
  private calcularDiarias(checkIn: Date, checkOut: Date): number {
    const diff = checkOut.getTime() - checkIn.getTime();
    return Math.ceil(diff / (1000 * 60 * 60 * 24));
  }
}

// Context que usa a strategy
@Injectable()
export class ReservasService {
  private calculoStrategy: ICalculoValorReservaStrategy;
  
  constructor(
    private readonly repository: IReservasRepository,
    private readonly quartosService: QuartosService
  ) {
    // Strategy padrão
    this.calculoStrategy = new CalculoValorPadraoStrategy();
  }
  
  setCalculoStrategy(strategy: ICalculoValorReservaStrategy): void {
    this.calculoStrategy = strategy;
  }
  
  async criarReserva(dto: CriarReservaDto): Promise<Reserva> {
    return await this.prisma.$transaction(async (tx) => {
      const quarto = await this.quartosService.buscarPorId(dto.quartoId);
      
      if (!quarto.isDisponivel()) {
        throw new ConflictException('Quarto indisponível');
      }
      
      // Calcular valor usando strategy atual
      const valorTotal = this.calculoStrategy.calcular(
        { 
          dataCheckIn: dto.dataCheckIn, 
          dataCheckOut: dto.dataCheckOut 
        } as Reserva,
        quarto
      );
      
      const reserva = ReservaFactory.criar({
        ...dto,
        valorTotal
      });
      
      const reservaCriada = await this.repository.create(reserva);
      await this.quartosService.alterarStatus(dto.quartoId, StatusQuarto.OCUPADO);
      
      return reservaCriada;
    });
  }
  
  // Método para definir strategy baseado em lógica de negócio
  private determinarStrategy(
    checkIn: Date,
    checkOut: Date
  ): ICalculoValorReservaStrategy {
    const diarias = this.calcularDiarias(checkIn, checkOut);
    
    // Alta temporada (exemplo: dezembro)
    if (this.isAltaTemporada(checkIn, checkOut)) {
      return new CalculoValorAltaTemporadaStrategy(1.5);
    }
    
    // Longa permanência
    if (diarias >= 7) {
      return new CalculoValorLongaPermanenciaStrategy();
    }
    
    // Padrão
    return new CalculoValorPadraoStrategy();
  }
  
  private isAltaTemporada(checkIn: Date, checkOut: Date): boolean {
    // Lógica para determinar alta temporada
    // Exemplo: dezembro, janeiro, julho
    const meses = [checkIn.getMonth(), checkOut.getMonth()];
    return meses.some(m => [11, 0, 6].includes(m));
  }
  
  private calcularDiarias(checkIn: Date, checkOut: Date): number {
    const diff = checkOut.getTime() - checkIn.getTime();
    return Math.ceil(diff / (1000 * 60 * 60 * 24));
  }
}
```

  ) {}

          reservaCriada.dataCheckOut
        )
      );
      
      return reservaCriada;
    });
  }
  
  async cancelarReserva(
    id: string,
    motivo?: string
  ): Promise<void> {
    return await this.prisma.$transaction(async (tx) => {
      const reserva = await this.repository.findById(id);
      
      if (!reserva) {
        throw new NotFoundException('Reserva não encontrada');
      }
      
      await this.repository.cancelar(id);

        'reserva.cancelada',
        new ReservaCanceladaEvent(
          reserva.id,
          reserva.quartoId,
          reserva.hospedeId,
          motivo
        )
      );
    });
  }

### 🟡 3.1 Chain of Responsibility

**Prioridade:** BAIXA  
**Complexidade:** MÉDIA  
**Impacto:** MÉDIO

**Aplicação:**
Sistema de validação em cadeia para reservas.

**Implementação:**

```typescript
interface IValidacaoReservaHandler {
  setNext(handler: IValidacaoReservaHandler): IValidacaoReservaHandler;
  handle(dto: CriarReservaDto): Promise<void>;
}

abstract class ValidacaoReservaHandler implements IValidacaoReservaHandler {
  private nextHandler: IValidacaoReservaHandler;
  
  setNext(handler: IValidacaoReservaHandler): IValidacaoReservaHandler {
    this.nextHandler = handler;
    return handler;
  }
  
  async handle(dto: CriarReservaDto): Promise<void> {
    await this.validate(dto);
    
    if (this.nextHandler) {
      await this.nextHandler.handle(dto);
    }
  }
  
  protected abstract validate(dto: CriarReservaDto): Promise<void>;
}

class ValidarQuartoExisteHandler extends ValidacaoReservaHandler {
  constructor(private quartosService: QuartosService) {
    super();
  }
  
  protected async validate(dto: CriarReservaDto): Promise<void> {
    const quarto = await this.quartosService.buscarPorId(dto.quartoId);
    if (!quarto) {
      throw new NotFoundException('Quarto não encontrado');
    }
  }
}

class ValidarQuartoDisponivelHandler extends ValidacaoReservaHandler {
  constructor(private quartosService: QuartosService) {
    super();
  }
  
  protected async validate(dto: CriarReservaDto): Promise<void> {
    const quarto = await this.quartosService.buscarPorId(dto.quartoId);
    if (!quarto.isDisponivel()) {
      throw new ConflictException('Quarto não está disponível');
    }
  }
}

class ValidarHospedeExisteHandler extends ValidacaoReservaHandler {
  constructor(private hospedesService: HospedesService) {
    super();
  }
  
  protected async validate(dto: CriarReservaDto): Promise<void> {
    const hospede = await this.hospedesService.buscarPorId(dto.hospedeId);
    if (!hospede) {
      throw new NotFoundException('Hóspede não encontrado');
    }
  }
}

class ValidarDatasHandler extends ValidacaoReservaHandler {
  protected async validate(dto: CriarReservaDto): Promise<void> {
    const checkIn = new Date(dto.dataCheckIn);
    const checkOut = new Date(dto.dataCheckOut);
    
    if (checkOut <= checkIn) {
      throw new BadRequestException(
        'Data de check-out deve ser posterior ao check-in'
      );
    }
    
    if (checkIn < new Date()) {
      throw new BadRequestException(
        'Data de check-in não pode ser no passado'
      );
    }
  }
}

// Uso no Service
@Injectable()
export class ReservasService {
  private validationChain: IValidacaoReservaHandler;
  
  constructor(
    private quartosService: QuartosService,
    private hospedesService: HospedesService
  ) {
    this.setupValidationChain();
  }
  
  private setupValidationChain(): void {
    const quartoExiste = new ValidarQuartoExisteHandler(this.quartosService);
    const quartoDisponivel = new ValidarQuartoDisponivelHandler(this.quartosService);
    const hospedeExiste = new ValidarHospedeExisteHandler(this.hospedesService);
    const datasValidas = new ValidarDatasHandler();
    
    quartoExiste
      .setNext(quartoDisponivel)
      .setNext(hospedeExiste)
      .setNext(datasValidas);
    
    this.validationChain = quartoExiste;
  }
  
  async criarReserva(dto: CriarReservaDto): Promise<Reserva> {
    // Executar cadeia de validação
    await this.validationChain.handle(dto);
    
    // Criar reserva
    return await this.prisma.$transaction(async (tx) => {
      // ...
    });
  }
}
```

- ✅ Fácil adicionar/remover validações
- ✅ Ordem de validação configurável

// Decorator de cache
class CachedQuartosRepository implements IQuartosRepository {
  constructor(
    private repository: IQuartosRepository,
    private cache: CacheService
  ) {}
  
  async findAll(filtros?: QuartoFilters): Promise<Quarto[]> {
    const cacheKey = `quartos:${JSON.stringify(filtros)}`;
    
    // Tentar buscar no cache
    const cached = await this.cache.get<Quarto[]>(cacheKey);
    if (cached) {
      return cached;
    }
    
    // Buscar no repository
    const quartos = await this.repository.findAll(filtros);
    
    // Armazenar no cache (TTL 5 minutos)
    await this.cache.set(cacheKey, quartos, 300);
    
    return quartos;
  }
  
  async findById(id: string): Promise<Quarto | null> {
    const cacheKey = `quarto:${id}`;
    
    const cached = await this.cache.get<Quarto>(cacheKey);
    if (cached) {
      return cached;
    }
    
    const quarto = await this.repository.findById(id);
    
    if (quarto) {
      await this.cache.set(cacheKey, quarto, 300);
    }
    
    return quarto;
  }
  
  // Invalidar cache ao modificar
  async create(quarto: Quarto): Promise<Quarto> {
    const created = await this.repository.create(quarto);
    await this.cache.invalidate('quartos:*');
    return created;
  }
  
  // ... outros métodos
}

        }

    });
  }
